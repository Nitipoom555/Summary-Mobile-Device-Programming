2. SQLITE DATABASE (ข้อมูลท้องถิ่น)

// เปิดหรือสร้างฐานข้อมูล
SQLiteDatabase db = this.openOrCreateDatabase("mydb", MODE_PRIVATE, null);
// หรือ: db = SQLiteDatabase.openDatabase(path, null, SQLiteDatabase.CREATE_IF_NECESSARY);

// สร้าง Table (ใช้วิธีรัน SQL ตรงๆ)
db.execSQL("CREATE TABLE tb1 (recID integer PRIMARY KEY autoincrement, name text, phone text)");

// INSERT (ใช้ ContentValues) แนะนำวิธีนี้
ContentValues vals = new ContentValues();
vals.put("name", "ABC"); 
vals.put("phone", "555-1010");
long rowId = db.insert("tb1", null, vals);

// UPDATE (อัปเดตข้อมูล)
ContentValues upd = new ContentValues();
upd.put("name", "Maria");
int recAffected = db.update("tb1", upd, "recID = ?", new String[]{"1"});

// DELETE (ลบข้อมูล)
db.delete("tb1", "recID > ? AND name = ?", new String[]{"5", "Maria"});

// QUERY & CURSOR (ดึงข้อมูลมาอ่าน)
Cursor c = db.rawQuery("SELECT * FROM tb1", null);
// หรือพารามิเตอร์: db.rawQuery("SELECT * FROM tb1 WHERE name=?", new String[]{"ABC"});
c.moveToPosition(-1); // ถอยไปจุดเริ่มก่อน
while(c.moveToNext()) {
  int id = c.getInt(0); // index คอลัมน์
  String name = c.getString(1); 
  // ใช้ชื่อคอลัมน์หา index ได้
  String phone = c.getString(c.getColumnIndex("phone"));
}
c.close();

// TRANSACTIONS (ป้องกัน Error ลบ/เขียนไม่สมบูรณ์)
db.beginTransaction();
try {
  // .. รัน SQL ..
  db.setTransactionSuccessful(); // ยืนยัน Commit
} catch (SQLiteException e) {
} finally { 
  db.endTransaction(); // คืนสถานะ
}

3. WEBVIEW & MIXED MODE

// อย่าลืม Permission ใน AndroidManifest.xml

WebView webview = (WebView) findViewById(R.id.webView1);

// โหลด HTML String ดิบๆ ลงไปเลย
String html = "<html><body><h1>Hello</h1></body></html>";
webview.loadData(html, "text/html", "UTF-8");

// การคุม Navigation ปุ่มถอย/เดินหน้า
if(webview.canGoBack()) webview.goBack();
if(webview.canGoForward()) webview.goForward();
webview.reload();
webview.clearCache(true);

4. CANVAS (วาดกราฟิก 2D)

// ต้องสร้างคลาสย่อย extends View แล้ว Override onDraw
@Override
protected void onDraw(Canvas canvas) {
  super.onDraw(canvas);
  Paint paint = new Paint();
  paint.setStyle(Paint.Style.FILL);
  paint.setColor(Color.parseColor("#CD5C5C"));
  
  // วาดวงกลม (พิกัด x, y, รัศมี, ออปเจกต์สี)
  canvas.drawCircle(150, 200, 50, paint);
  
  // วาดสี่เหลี่ยมขอบมน (ซ้าย, บน, กว้าง, สูง) , โค้ง x, โค้ง y
  canvas.drawRoundRect(new RectF(20,20,100,100), 20, 20, paint);
  
  // หมุนผืนผ้าใบก่อนวาดข้อความ
  canvas.rotate(-45);
  canvas.drawText("Tutorial", 40, 180, paint);
  canvas.restore();
}

5. ANIMATION (ฟิสิกส์ & เฟรม)

     .setMinValue(0)
     .setMaxValue(1000)
     .setFriction(0.1f)
     .start();

6. ALARM MANAGER (ตั้งเวลาทำงาน)

Intent i = new Intent(this, AlarmReceiver.class);
// สั่งให้ PendingIntent ไปปลุก BroadcastReceiver
PendingIntent pIntent = PendingIntent.getBroadcast(this, 0, i, 0);

AlarmManager alarm = (AlarmManager) getSystemService(ALARM_SERVICE);
// สั่งทำซ้ำๆ ทุก 10 วิ (10000ms) อิงตามเวลาจริงโลก
alarm.setRepeating(AlarmManager.RTC_WAKEUP, 
    cal.getTimeInMillis(), 10000, pIntent);

// การยกเลิก
alarm.cancel(pIntent);
